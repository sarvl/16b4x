[TERMINOLOGY]
	WORD
		2B wide 
		smallest addressable unit of data
	REGISTER
		general purpose register
		accessible by all instructions requiring register
		shortened to R or R[]
	AUXILIARY REGISTER
		special purpose register 
		can be modified with XWR and read with XRD
		shortened to X or X[]
	CONTROL REGISTER
		special purpose register for control data and info of specific cpu features
		can be modified with CWR and read with CRD
		can only be modified in privileged mode
		depending on id, may be readable in unprivileged mode
		shortened to C or C[]
	IMPLEMENTATION REGISTER
		register used by implementation
		no way to access except via special instructions
		shortened to I or I[]
	FRAME
		page in physical memory
	MEMORY
		general volatile storage, does not persist across reboots
		shortened to M or M[]

[OVERVIEW]
	[OVERVIEW:GENERAL]
		version:            0 
		codename:           16b40
		this doc revision:  0 

		Before reading this document (especially sections about IO), it is important to be aware how general 16b4x system is supposed to work
		check system.txt for that info
		
		Future ISA versions will be backwards compatible unless a bug is found 

		Extensions may be eventually merged into base ISA, corresponding feature bit (found in corresponding CR) then is permanently '1'

		Negative numbers are represented in Two's complement form

		Currently the ISA is in release candidate state

[INSTRUCTION_ENCODING]
	[INSTRUCTION_ENCODING:GENERAL]
		argument order is the same as in regular instruction description
		each argument is marked with where it goes 
		for example MWR S D means that 00000SSS DDD01101 should be used
		                                  arg0^ ^arg1
		            MRD D S means that 00000SSS DDD01100 should be used
		                                  arg1^ ^arg0
		
		unused/ignored fields are required to be 0 


	[INSTRUCTION_ENCODING:IMMEDIATES]
		immediates are encoded from most significant to least significant bit

		immediates are sign extended by 
			JMP, JCC, CAL, LOP
		all other instructions zero extend their immediate

		if XWR UI precedes an instruction with an immediate then 
			lower 8 bits of UI are shifted to the left 
			and merged with LOWER 8 bits of immediate
			which means that JMP and CAL will act as if they have 8 bit immediate, not 11
			immediate is always zero extended

	[INSTRUCTION_ENCODING:SO]
		[INSTRUCTION_ENCODING:SO:LI]
			OOOOOIII IIIIIIII
				O - opcode
				I - imm11

			used by:
				CAL I, JMP I
		
		[INSTRUCTION_ENCODING:SO:JC]
			OOOOOCCC IIIIIIII
				O - opcode
				I - imm8
				C - condition codes

			used by:
				JCC C I
				^ condition is determined based on instruction
				v0 is x2000 ; v1 is x2800

		[INSTRUCTION_ENCODING:SO:MC]
			OOOOOSSS DDD0CCC0
				O - opcode
				S - source register id
				D - destination register id
				C - condition codes
				0 - MUST be 0
			
			used by:
				MCC D S
				^ condition is determined based on instruction
				v0 is xB000 ; v1 is xB800

		[INSTRUCTION_ENCODING:SO:RI]
			OOOOOIII RRRIIIII
				O - opcode
				I - imm8
				R - register

			used by:
				ADD R I, AND R I, CMP R I
				MLU R I, MOV R I, MRD R I, MWR I R, ORR R I,
				PRD R I, PWR I R, SHL R I, SHR R I, SRD R I,
				SUB R I, SWR I R, TST R I, XOR R I, XWR R I

		[INSTRUCTION_ENCODING:SO:RR]
			00000SSS DDDOOOOO
				0 - MUST be set to 0
				S - source register
				D - destination register
				O - opcode

			used by:
				ADD D S, AND D S, ANN D S, CMP D S, CRD D S,
				CWR S D, DVU D S, DVS D S, MLU D S, MLS D S,
				MOV D S, MRD D S, MWR S D, ORR D S, PRD D S, 
				PWR S D, SHL D S, SHR D S, SUB D S, TST D S, 
				XOR D S, XRD D S, XWR D S

				SCC D
				condition to check (ccc) is encoded in S
				v0 is x0001
				v1 is x0017

	[INSTRUCTION_ENCODING:MO]
		[INSTRUCTION_ENCODING:MO:I]
			OOOOOOOO IIIIIIII	
				O - opcode 
				I - imm8 

			used by
				INT I, LOP I

		[INSTRUCTION_ENCODING:MO:RI]
			OOOOOOOO RRRIIIII
				O - opcode
				R - register id 
				I - imm5 
				
			used by:
				CRD R I, CWR I R

	[INSTRUCTION_ENCODING:LO]
		[INSTRUCTION_ENCODING:LO:R]
			OOOOOOOO RRROOOOO	
				O - opcode 
				R - register 

			used by:
				CAL R, NEG R, NOT R, POP R,
				PRF R, PSH R, RNG R  

				HLT, IRT, NOP, RET
				^ these instructions ignore R field

[REGISTERS]
	[REGISTERS:INTERNAL]
		there are 8 registers available for use in most of the instructions
		denoted by R[x] in instruction description 
		
		0 - R0 ; General Purpose Register 0
		1 - R1 ; General Purpose Register 1
		2 - R2 ; General Purpose Register 2
		3 - R3 ; General Purpose Register 3
		4 - R4 ; General Purpose Register 4
		5 - R5 ; General Purpose Register 5
		6 - R6 ; General Purpose Register 6
		7 - R7 ; General Purpose Register 7

	[REGISTERS:AUXILIARY]
		[REGISTERS:AUXILIARY:LIST]
			0 IP Instruction Pointer
			1 LC Loop Count
			2 UI Upper Immediate (during Write, read is reserved)
			3 SP Stack Pointer
			4 FL FLags
			5 AR Arithmetic Register
			6 currently unused, reserved
			7 currently unused, reserved

		[REGISTERS:AUXILIARY:IP] Instruction Pointer
			holds address of NEXT instruction
			writing to this register is equivalent to unconditional jump
				this is the only way to perform **jump** to absolute address
				it is possible however to perform CAL register which calls absolute address

			
			each cycle:
			if instruction was any of: INT, JMP, CAL, RET, IRT
			or if instruction was JCC or LOP and the jump occurred 
			or if instruction was XWR IP, /*something*/
			then value of IP is changed by that instruction
			otherwise, it is incremented by 1 

		[REGISTERS:AUXILIARY:LC] Loop Count
			holds loop count used by LOP instruction
			
			provides quick way to implement loops by reducing number of instructions to execute

		[REGISTERS:AUXILIARY:UI] Upper Immediate
			this register cannot be read from directly

			extends range of immediate by adding additional 8 bits
			note that some instructions may completely ignore extended part 
			see [INSTRUCTION_ENCODING:IMMEDIATE] for details

			each cycle:
			if XWR UI X occurs: UI <-- X
			otherwise         : UI <-- 0

		[REGISTERS:AUXILIARY:SP] Stack Pointer
			used implicitly by POP, PSH, CAL, RET, SRD, SWR

			see [MEMORY:STACK]


		[REGISTERS:AUXILIARY:FL] FLags register
			contains 5 flag bits
			 00000000 000DSOCZ
			 D - division by zero                  , defined for DVS, DVU
			 S - most significant bit (sign) = 1
			 O - carry out of MSb /= carry into MSb, defined for ADD, SUB, CMP, DVS, DVU
			 C - carry out of MSb = 1              , defined for ADD, SUB, CMP
			 Z - all bits = 0 

			 ALL flags may be modified by ALL instructions modifying flags

			 S and Z are always defined

			 D flag and div by zero interrupts are set to 1 simultaneously, this provides flexibility to the software to pick preffered method of handling the error
			  to use flag, silence the interrupt
			  to use interrupt, simply ignore the flag
			 if D extension is not present, this flag is always set to 0


			 on any write, unused bits MUST be set to 0 
			 otherwise the behavior is undefined

		[REGISTERS:AUXILIARY:AR] Arithmetic Register
			Present only if [EXTENSIONS:A] is present

			used to extend the range of available operations for DVS, DVU, MLS, and MLU
			for details check behavior of each instruction

	[REGISTERS:CONTROL]
		[REGISTERS:CONTROL:GENERAL]
			access depends on range, write is always privileged
			not used (not listed) registers are reserved and MUST NOT be used

			x0000 to x3FFF -      architectural,   privileged read	
			x4000 to x7FFF -      architectural, unprivileged read	
			x8000 to xBFFF - microarchitectural,   privileged read	
			xC000 to xFFFF - microarchitectural, unprivileged read	

			microarchitectural registers are listed in corresponding manual
			first 32 architectural privileged read registers can be accessed directly, others require putting address into register

			bits are indexed from least to most significant with hexadecimal digits

			when it does not follow from the descirption, value in parentheses indicates starting value
			 undefined means that the value can be whatever and running program must not depend on it

		[REGISTERS:CONTROL:LIST]
			x0000 : PID of running program                                     (x0000)
			x0001 : PT address                                                 (undefined)
			x0002 : privilege level settings 0                                 (xFFFF)
			         5 - [I] interrupt control
			         4 - [V] virtual memory control
			         3 - [P] protected memory access
			         2 - [C] control register access
			         1 - [O] IO access
			         0 - [R] processor control

					 the rest should be set to 0 for the least privileged mode, and 1 for the most privileged mode

			x0003 : privilege level settings 1                                 (xFFFF)

			         currently not used

					 all bits should be set to 0 for the least privileged mode, and 1 for the most privileged mode

			x0004 : interrupt enabled 0  see [INTERRUPTS:CONTROL]                                       (x0000)
			x0005 : interrupt enabled 1  see [INTERRUPTS:CONTROL]                                       (x0000)

			x000E : state register
			        0 - interrupts enabled

					other bits should be set to 0 for future compatibility

			x000F : state update 
				write of particular bits causes update of machine state
				note that write to this register may take many cycles, depending on how much HW has to do 

					 1 - flush TLB
					 0 - IHT reload 

					writing '1' to any other bit may cause unexpected behavior in the future

			x0010 : interrupt ID        see [INTERRUPTS]                                                (undefined)
			x0011 : interrupt saved IP                                                                  (undefined)
			x0012 : saved CR[x0002]                                                                     (undefined)
			x0013 : saved CR[x0003]                                                                     (undefined)
			x0014 : saved UI                                                                            (undefined)
			x0015 : saved FL                                                                            (undefined)

			x0016 : reserved                                                                            (undefined)
			x0017 : reserved                                                                            (undefined)

			x0018 : interrupt info                                                                      (undefined)
			x0019 : interrupt info                                                                      (undefined)
			x001A : reserved                                                                            (undefined)
			     ...
			x001F : reserved                                                                            (undefined)


			x1000 : ISA info
			         MSB - version ; LSB - revision
			x1001 : microarchitecture ID 
			x1002 : catalyst info
			         MSB - version ; LSB - revision

			0x1010 : substrate info register 0 - available frame count in physical RAM, at least 32     (undefined)
			0x1011 : substrate info register 1 -                                                        (undefined)
			      ...                                                                                   (undefined)
			0X101F : substrate info register F -                                                        (undefined)

			x2000 : PAF 0 see [CPU_FEATURES]
			x2001 : guaranteed to return 0
			     ...
			x200F : guaranteed to return 0

			x5000 : UAF 0 see [CPU_FEATURES]
			x5001 : guaranteed to return 0
			     ...
			x500F : guaranteed to return 0

	[REGISTERS:IMPLEMENTATION]
		since these registers are implementation specific, they are not described here 
		check corresponding microarchitecture manual 

[CPU_FEATURES]
	[CPU_FEATURES:GENERAL]
		Features are part of CRs, see [REGISTERS:CONTROL]

		These registers cannot be altered

		PAF -   Privileged      Architectural Feature register
		PUF -   Privileged Microarchitectural Feature register
		UAF - Unprivileged      Architectural Feature register
		UUF - Unprivileged Microarchitectural Feature register

		microarchitectural registers are described in microarchitecture manual
		
		[CPU_FEATURES:LIST] lists what each bit means for given register, 0 is LSb, F is MSb
		unlisted bits are guaranteed to be 0 for future compatiblity

	[CPU_FEATURES:LIST]
		PAF 0 
			0 - Fast Boot, RAM already contains data from lowest 64kiW of drive
			    primarily useful for simulation environments

		UAF 0

			1 - [R] Random extension
	

[BOOT]
	before any execution happens, control registers are initialized according to [REGISTERS:CONTROL:LIST]

	immediattely after that processor reads catalyst version and puts it into CR[x1002]
	 note that there are few more registers with control information (substrate info registers 0 until F)
	 it is the job of software bootloader or OS to initialize them properly according to this document

	then IO_INIT command is issued to the Catalyst,
	once the catalyst is ready 
	 machine copies first 512kiW from persistent storage to memory starting from address x0000
	 by setting P[x000E] to address and reading data from P[x000F], for each copied word

	initially TLB maps entire virtual address space 1:1 into physical address space
	that is, memory sees exactly the same addresses as programs
	physical addresses that are above maximum 16bit unsigned integer value are not accessible until virtual memory is configured

	And finally execution begins from x0000 in fully privileged mode with interrupts disabled

[MEMORY]
	[MEMORY:GENERAL]
		memory is word addressable, where each word is 2Bytes
		
		at least 64kiW (128kiB) is available

	[MEMORY:STACK]
		managed implicitly by POP, PSH, CAL, RET
		SP points to the top of the stack
			NOT to where next value should go
		 explicit memory accesses can also be used but they are generally less efficient

		PSH first decrements SP THEN puts value
		POP first gets value THEN increments SP

		w.r.t. stack, CAL acts like PSH
		w.r.t. stack, RET acts like POP
		
		local variables can be easily accessed with SRD and SWR

	[MEMORY:VIRTUAL]
		each page is 2kiW in size, thus full PT stores mapping for 32 pages

		on boot, paging is enabled and virtual addreses are translated 1:1 into physical
		it is guaranteed that TLB can always contain address space of currently running program
		so as long as PID is not changed, there is no need for OS to create page table

		note that if more than 64kiW of memory is provided, it is inaccessible untill proper paging is set up

		PT can be stored anywhere as long as its PHYSICAL address fits into 16bits
		PHYSICAL address of PT is then saved into CR[x0001]
		when TLB miss occurs, that address is used to access PT
			see [MEMORY:VIRTUAL:PAGE_TABLE:LAYOUT] for format

		it is impossible to disable paging

		when page is written to for the first time (its dirty bit is 0) then on write CPU sets dirty bit in page table to 1

		[MEMORY:VIRTUAL:PID]
			always located in CR[x0000]

			CPU may cache PT of more than 1 process at once 
			it is therefore necessary for CPU to distinguish different processes
			pid is checked regardless of privilege setting
			
		[MEMORY:VIRTUAL:PAGE_TABLE]
			[MEMORY:VIRTUAL:PAGE_TABLE:LAYOUT]
				each entry is 1W (16b) 

				32 entries, 1 for each page, ordered 
				[present] [protected] [dirty] [read] [write] [execute] [frame #]
					  1b          1b      1b     1b      1b        1b       10b 

				if there is no frame with id provided, the behavior is undefined
				OS should make sure that a particular page is available by checking contents of CR[x1010]
				 it is guaranteed that at least 32 frames are available, indexed from 0
				 the mapping is always continuous

			[MEMORY:VIRTUAL:PAGE_TABLE:UPDATE]
				OS MUST flush TLB (CR[x000F]) after PT update is finished AND proper PID is set 
				 otherwise HW may use old mapping
				 this removes all entries with pid matching current pid from TLB
				 however it leaves other entries unaffected, this lets HW cache entries of more than 1 entry at a time, possibly speeding up context switches


		[MEMORY:VIRTUAL:PAGE_FAULT]	
			if program attempts to access page that is not present in page table, page fault interrupt is raised
			instruction can be restarted if corresponding entry is put into PT

		[MEMORY:VIRTUAL:PROTECTION]
			'protected' bit is used to verify privilege level, if it is set, then the page entry is available only when [P] privilege is enabled
			see [PRIVILEGE_LEVEL:LIST]
			if process tries to access protected page without having the necessary privilege, privilege violation interrupt is raised

			each page entry contains bits for read, write, execute access
			if process tries to do any action which it cannot based on page table entry
				permission violation interrupt is raised 


[PRIVILEGE_LEVEL]
	[PRIVILEGE_LEVEL:GENERAL]
		each category of features gets its own switch in CR[x0002] and CR[x0003], if corresponding bit is `1` then that feature is enabled 
		 see [REGISTERS:CONTROL:LIST]
		therefore to achieve fully unprivileged mode, CR[x0002] and CR[x0003] should be set to 0
		to achieve fully privileged mode, they should be set to xFFFF

		currently only CR[x0002] has any flags, but CR[x0003] is reserved for the future
		
	[PRIVILEGE_LEVEL:BOOT]
		on boot, CR[x0002] and CR[x0003] are set to xFFFF - meaning full privilege is available
		this can later be changed provided Control Register Access (C) privilege is enabled

	[PRIVILEGE_LEVEL:LIST]
		[R] processor control
			general protection switch used to control instructions that don't fit any other category
			currently controls only HLT instruction

		[O] IO access
			protection level used to access IO with PRD and PWR
			see [IO]

		[C] control register access
			ability to read/write all control registers
			see [REGISTERS:CONTROL]

		[P] protected memory access
			ability to access pages with Protected bit set 
			see [MEMORY:VIRTUAL]

		[V] virtual memory control
			ability to modify the Page Table
			see [MEMORY:VIRTUAL]

		[I] interrupt control
			ability to send all interrupts and to change IHT and IHR
			see [INTERRUPTS]

			
[IO]
	communication with different devices is performed via ports or interrupts
	interrupts are preferred method of communication for start of communication and rarely occuring events
	ports should be used in all other scenarios 
	see [INTERRUPTS]

	if communication is timing sensitive, interrupts should be disabled

	for description of communication see Catalyst documentation	
			
[INTERRUPTS]
	[INTERRUPTS:GENERAL]
		while there is distinction between exceptions, interrupts, faults, and traps
		 this ISA treats them the same and thus they are collectively referred to as interrupts
		 whenever distinction is needed, terms 'external interrupt' for interrupt and 'internal interrupt' for exceptions, faults, or traps are used  
		
		INT instruction can cause any kind of interrupt
		however if the interrupt is in range 0 to 127 then it can be run only with Interrupt Control privilege (see [PRIVILEGE_LEVEL])
		system calls can be implemented via interrupts 128 to 255

		if INT raises any HW defined interrupt, contents of CRs reserved for interrupts are undefined
		
		initially, interrupts are fully surpressed, this can be changed via CR[x000F]
		 note that this also means that NMI are fully surpressed, this is necessary to surpress potential interrupts under interrupts
	
	[INTERRUPTS:LIST]
		  0 to  31 - internal       , non maskable
		 32 to  63 - internal       , maskable
		 64 to 127 - IO             , maskable through Catalyst
		128 to 255 - available to SW, non maskable
			OS can set them up for various syscalls 

		internal interrupts that are not listed are reserved 

		  0 - reserved 
		  ...
		  7 - reserved 
		  8 - PGF : page fault
		  9 - PRV : privilege violation
		 10 - PEV : permission violation (unallowed read/write/execute)
		 11 - INS : instruction not supported
		 12 - INA : instruction invalid argument

		 32 - TMR : timer
		 33 - DIV : division by 0  

	[INTERRUPTS:ORDERING]
		the lowest numbered interrupt always takes precedence while the other is delayed
		 it is possible that after first interrupt is handled, the other is not needed, in that case the second one stops being asserted

	[INTERRUPTS:PRECISE]
		all interrupts caused by INT instruction happen precisely AFTER INT but BEFORE next instruction
		
		list of internal interrupts that happen between instruction causing interrupt and next instruction 
			PGF : page fault
			PRV : privilege violation
			PEV : permission violation (unallowed read/write/execute)
			INS : instruction not supported
			DIV : division by 0  

		all other may occur on ANY instruction in the instruction stream 
		this is to simplify HW and allow for faster execution

	[INTERRUPTS:HANDLER_TABLE]
		IHT is 256x1word table where each entry is a PHYSICAL address of interrupt handler
		whenever interrupt with id N occurs, HW looks up Nth entry and jumps to address stored there

		IHT is specified by setting up table at VIRTUAL address x0100 then updating HW state with CR[x000F] (see [REGISTERS:CONTROL])
		HW loads IHT to internal registers and IHT in memory that was setup by SW is not needed anymore and is free to be removed
		it is not possible to modify only single entry 

	[INTERRUPTS:RAISED]
		source of interrupt does not matter, basic handling looks the same 
		it may happen that several interrupts are to be raised, in which case, the lowest number takes priority
		
		when interrupt occurs during execution of instruction, that instruction completes THEN 
			interrupt is raised immediately
			interrupts are disabled
			necessary data is saved into CR and internal registers (see below)
			privilege level is raised to fully privileged
			jump into interrupt handler occurs and execution resumes

		saved data
			CR[x0010] = interrupt id
			CR[x0011] = address of instruction
			             unless specified otherwise, instruction to be executed AFTER interrupt causing one
			CR[x0012] = saved CR[x0002]
			CR[x0013] = saved CR[x0003]
			CR[x0014] = saved upper immediate
			CR[x0015] = saved flags

			CR[x0018...x0019] = interrupt dependent info 


		if an interrupt is to be raised, but the interrupts are blocked
			instruction pointer increases as regular
			registers affected by the instruction may be updated 
			no external state changes - meaning no command is issued to IO and no memory is updated

	[INTERRUPTS:HANDLED]
		interrupts can, but dont have to, be finished with IRT 
		 IRT is just a convienience instruction to quickly finish handling interupt and return to the execution

		during IRT
			privilege level is read from CR[x0012] and CR[x0013] to CR[x0002] and CR[x0003] respectively
			UI is restored from CR[x0014]
			FL is restored from CR[x0015]
			IP is restored from CR[x0011]
			interrupts are reenabled 

	[INTERRUPTS:CONTROL]
		C[x0004 to x0005] can be used to enable interrupts in range 32 to 63 
		   0 to  31 are non maskable interrupts, they MUST be handled to keep machine running
		  64 to 127 are IO interrupts, to configure them use Catalyst
		 128 to 255 are syscalls which are not maskable

		interrupt 32 is enabled by C[x0004] bit  0
		          ...
		interrupt 47 is enabled by C[x0004] bit 15
		interrupt 48 is enabled by C[x0005] bit  0
		          ...
		interrupt 63 is enabled by C[x0005] bit 15

		
	[INTERRUPTS:HANDLER]
		there is no limit on interrupt handler length
		it is however advised that interrupt handlers are very short because they block other interrupts
		IHT entries can overlap like cases in switch statements in C
		
		exact behavior depends on interrupt

		[INTERRUPTS:HANDLER:PAGE_FAULT]
			CR[x0018] = address that caused page fault
			CR[x0019] = 0

			generated if there is no valid mapping for address that CPU tries to access
			state update is surpressed
			
			CR[x0011] is address of instruction that caused page fault, NOT the next one
			 instruction fetch is considered part of new instruction, not preceding one

		[INTERRUPTS:HANDLER:PRIVILEGE_VIOLATION]
			CR[x0018] = bit n corresponds to missing privilege n in register CR[x0002]
			CR[x0019] = bin n corresponds to missing privilege n in register CR[x0003]
			
			generated if CPU tries to execute instruction requiring privilege not possesed

			for example if CPU tries to execute HLT instruction, CR[x0018] = 1 and CR[x0019] = 0

		[INTERRUPTS:HANDLER:PERMISSION_VIOLATION]
			CR[x0018] = address that CPU tried to access in some way
			CR[x0019] = type of violation
				x0001 - read
				x0002 - write
				x0004 - execute

			generated if CPU tries to access address in illegal way

		[INTERRUPTS:HANDLER:INSTRUCTION_NOT_SUPPORTED]
			CR[x0018] = data that CPU tried to execute 
			CR[x0019] = 0

			generated if byte sequence is not a valid instruction or requires extension
			note that due to future extentions, it is impossible for CPU to distinguish between these two

		[INTERRUPTS:HANDLER:INSTRUCTION_INVALID_ARGUMENT]
			CR[x0018] = data that CPU tried to execute 
			CR[x0019] = 0

			generated if byte sequence is a valid instruction but contains invalid argument
			for example scc v1 with ccc=b111
		
		[INTERRUPTS:HANDLER:TIMER]
			CR[x0018] = 0
			CR[x0019] = 0
			
			nothing needs to be done, can be safely ignored
			but OS probably should utilize it to keep track of passed time

		[INTERRUPTS:HANDLER:DIVISION_BY_0]
			CR[x0018] = 0
			CR[x0019] = 0

			can be generated only if [EXTENSIONS:D] is available
			generated when divisor = 0 

		[INTERRUPTS:HANDLER:IO_INTERRUPT]
			CR[x0018] = 0
			CR[x0019] = 0

			generated by IO that requires attention, the exact action depends on interrupt ID and device located there

		[INTERRUPTS:HANDLER:SOFTWARE_INTERRUPT]
			CR[x0018] = 0
			CR[x0019] = 0

			generated by INT instruction, usually to perform system call

			JNZ JNE     => v1 ; ccc = 010 => Z = 0            ; no zero flag / not equal

			JSS         => v1 ; ccc = 100 => S = 1            ; sign flag

			MCC MAE MAZ => v0 ; ccc = 010 => C = 1            ; carry flag / unsigned above or equal
			MGE MGZ     => v0 ; ccc = 011 =>           S  = O ; signed greater or equal
			MGG         => v0 ; ccc = 100 => Z = 0 AND S  = O ; signed greater
			MLE MLZ     => v0 ; ccc = 101 => Z = 1 OR  S /= O ; signed less or equal
			MLL         => v0 ; ccc = 110 =>           S /= O ; signed less
			MNC MBB     => v0 ; ccc = 111 => C = 0            ; no carry flag / unsigned below
			MNO         => v1 ; ccc = 000 => O = 0            ; no overflow flag
			MNS         => v1 ; ccc = 001 => S = 0            ; no sign flag
			MNZ MNE     => v1 ; ccc = 010 => Z = 0            ; no zero flag / not equal
			MOO         => v1 ; ccc = 011 => O = 1            ; overflow flag
			MSS         => v1 ; ccc = 100 => S = 1            ; sign flag
			MZZ MEE     => v1 ; ccc = 101 => Z = 1            ; zero flag / equal

			v1 with ccc = 110 and ccc = 111 are reserved
			their usage raises INA interrupt

		auxiliary registers affected:
			-
		required extensions:
			-
		privilege level:
			-
